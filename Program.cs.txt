using System.Text;

string dir = args.Length > 0 ? args[0] : Directory.GetCurrentDirectory();
string csv = Path.Combine(dir, "result.csv");

if (!Directory.Exists(dir))
{
    Console.WriteLine($"Каталог не найден: {dir}");
    return;
}

var files = Directory.GetFiles(dir, "*.txt", SearchOption.AllDirectories);
if (files.Length == 0)
{
    Console.WriteLine("Файлы .txt не найдены");
    return;
}

Console.WriteLine($"Найдено {files.Length} файлов, обрабатываю...");

var sb = new StringBuilder();
sb.AppendLine("\"Файл\",\"Содержимое (изменено)\"");

foreach (var file in files)
{
    try
    {
        string text = File.ReadAllText(file);
        string modified = text.ToUpperInvariant();
        
        string escaped = modified.Replace("\"", "\"\"")
                                 .Replace("\n", "\\n")
                                 .Replace("\r", "\\r");
        
        sb.AppendLine($"\"{Path.GetFileName(file)}\",\"{escaped}\"");
        Console.WriteLine($"OK: {Path.GetFileName(file)}");
    }
    catch (Exception ex)
    {
        Console.WriteLine($"Ошибка: {Path.GetFileName(file)} - {ex.Message}");
    }
}

File.WriteAllText(csv, sb.ToString(), Encoding.UTF8);
Console.WriteLine($"\nГотово! Результат: {csv}");
